# Пример файла с прокси
# Формат: host:port:username:password
# Один прокси на строку

# HTTP/HTTPS прокси
proxy1.example.com:8080:user1:pass1
192.168.1.100:3128:admin:secret123

# SOCKS5 прокси
socks5.example.com:1080:myuser:mypass

# Без авторизации (оставить username:password пустыми)
proxy2.example.com:8888::

# Комментарии начинаются с #
# Пустые строки игнорируются

# Прокси будут автоматически распределены между аккаунтами
# Один аккаунт = один прокси (круговое распределение)
